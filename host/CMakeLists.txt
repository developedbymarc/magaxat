cmake_minimum_required(VERSION 3.28)

project(Magaxat-Host LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(MAGAXAT_METADATA_TOML "${CMAKE_CURRENT_LIST_DIR}/../metadata.toml")
set(MAGAXAT_METADATA_SCRIPT "${CMAKE_CURRENT_LIST_DIR}/cmake/generate_metadata.py")
set(MAGAXAT_METADATA_TEMPLATE "${CMAKE_CURRENT_LIST_DIR}/cmake/metadata.hpp.in")
set(MAGAXAT_METADATA_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(MAGAXAT_METADATA_CMAKE "${MAGAXAT_METADATA_BUILD_DIR}/metadata.cmake")
set(MAGAXAT_METADATA_INCLUDE_DIR "${MAGAXAT_METADATA_BUILD_DIR}/include")

set_property(
    DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${MAGAXAT_METADATA_TOML}"
    "${MAGAXAT_METADATA_SCRIPT}"
    "${MAGAXAT_METADATA_TEMPLATE}"
)

execute_process(
    COMMAND
        "${Python3_EXECUTABLE}"
        "${MAGAXAT_METADATA_SCRIPT}"
        "${MAGAXAT_METADATA_TOML}"
        "${MAGAXAT_METADATA_CMAKE}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/.."
    RESULT_VARIABLE MAGAXAT_METADATA_RESULT
    OUTPUT_VARIABLE MAGAXAT_METADATA_OUTPUT
    ERROR_VARIABLE MAGAXAT_METADATA_ERROR
)

if(NOT MAGAXAT_METADATA_RESULT EQUAL 0)
    message(
        FATAL_ERROR
        "Failed to generate CMake metadata:\n${MAGAXAT_METADATA_ERROR}"
    )
endif()

include("${MAGAXAT_METADATA_CMAKE}")
configure_file(
    "${MAGAXAT_METADATA_TEMPLATE}"
    "${MAGAXAT_METADATA_INCLUDE_DIR}/metadata.hpp"
    @ONLY
)

add_subdirectory(src)
